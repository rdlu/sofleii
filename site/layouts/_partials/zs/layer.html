{{- /* One layer of the keymap, drawn on the board's physical layout. Built on every site build from the keymap and keymap-drawer's parse of it (see zs/layer-data.html), so it can't drift from the firmware. Hover or focus a key for its position and raw binding. Takes (dict "name" "NAV"), plus optional "legend" false (no key-style legend) and "caption" (a one-line caption shown instead of the legend). Used by the `layer` shortcode and the print page (keymap-print.html). */ -}} {{- $name := .name -}} {{- $d := partialCached "zs/layer-data.html" . "layer-data" -}} {{- $keys := index $d.layers $name -}} {{- if not $keys }}{{ errorf "layer: no layer %q in the keymap (have %s)" $name (delimit $d.order ", ") }}{{ end -}} {{- $num := 0 }}{{ range $i, $n := $d.order }}{{ if eq $n $name }}{{ $num = $i }}{{ end }}{{ end -}} {{- $b := $d.box -}} {{- $over := index hugo.Data.keymap_overrides $name | default dict -}}
